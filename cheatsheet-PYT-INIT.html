Programmation orientée objet

from abc import ABC, abstractmethod

class Animal(ABC):
    espece = "inconnue"               # attribut de classe / static
    def __init__(self, nom: str):
        self._nom = nom              # protégé
    @abstractmethod
    def crier(self): pass

class Chien(Animal):
    def crier(self):
        print(f"{self._nom} aboie")

class Generique[T]:
    def __init__(self, v: T):
        self.valeur = v
